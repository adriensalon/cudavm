cmake_minimum_required(VERSION 3.24)

function(cudavm_merge_static_libraries target)
    set(root "${CMAKE_CURRENT_BINARY_DIR}/CMakeFiles/${target}_archives")
    set(merged "$<TARGET_FILE:${target}>.merged")
    if(WIN32)
        get_filename_component(tool_directory "${CMAKE_AR}" DIRECTORY)
        find_program(archive_tool NAMES llvm-lib lib HINTS "${tool_directory}" NO_CACHE REQUIRED)
        set(command_file "${root}/$<CONFIG>.rsp")
        set(contents "/NOLOGO\n/OUT:\"${merged}\"\n\"$<TARGET_FILE:${target}>\"\n")
        foreach(archive IN LISTS ARGN)
            string(APPEND contents "\"${archive}\"\n")
        endforeach()
        set(merge_command "${archive_tool}" "@${command_file}")
    else()
        set(command_file "${root}/$<CONFIG>.mri")
        set(contents "CREATE \"${merged}\"\nADDLIB \"$<TARGET_FILE:${target}>\"\n")
        foreach(archive IN LISTS ARGN)
            string(APPEND contents "ADDLIB \"${archive}\"\n")
        endforeach()
        string(APPEND contents "SAVE\nEND\n")
        set(merge_command sh -c "\"$1\" -M < \"$2\"" sh "${CMAKE_AR}" "${command_file}")
    endif()
    file(GENERATE OUTPUT "${command_file}" CONTENT "${contents}")
    set(member "${root}/dependencies_$<CONFIG>.cpp")
    add_custom_command(OUTPUT "${member}"
        COMMAND "${CMAKE_COMMAND}" -E touch "${member}"
        DEPENDS ${ARGN} "${command_file}" "${CMAKE_CURRENT_FUNCTION_LIST_FILE}" VERBATIM)
    target_sources(${target} PRIVATE "${member}")
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${merge_command}
        COMMAND "${CMAKE_COMMAND}" -E rename "${merged}" "$<TARGET_FILE:${target}>" VERBATIM)
endfunction()

function(cudavm_embed_file path output symbol)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" embedded_contents)
    string(FIND "${embedded_contents}" ")cudavm_resource\"" delimiter_collision)
    if(NOT delimiter_collision EQUAL -1)
        message(FATAL_ERROR "Embedded source contains the raw-string delimiter: ${path}")
    endif()
    set(embedded_symbol "${symbol}")
    file(CONFIGURE OUTPUT "${output}" CONTENT [[static constexpr char @embedded_symbol@[] = R"cudavm_resource(@embedded_contents@)cudavm_resource";
]] @ONLY)
endfunction()

function(_cudavm_prepare_target target)
    get_target_property(kind ${target} TYPE)
    get_target_property(no_deploy ${target} CUDAVM_NO_DEPLOY)
    set(deploy FALSE)
    if(WIN32 AND NOT no_deploy AND kind MATCHES "^(EXECUTABLE|SHARED_LIBRARY|MODULE_LIBRARY)$")
        set(deploy TRUE)
    endif()
    set(queue ${target})
    set(visited)
    set(files)
    while(queue)
        list(POP_FRONT queue library)
        string(REGEX REPLACE "^\\$<LINK_ONLY:([^>]+)>$" "\\1" library "${library}")
        if(NOT TARGET "${library}")
            continue()
        endif()
        get_target_property(owner ${library} ALIASED_TARGET)
        if(owner)
            set(library ${owner})
        endif()
        if(library IN_LIST visited)
            continue()
        endif()
        list(APPEND visited ${library})
        get_target_property(crt ${library} INTERFACE_CUDAVM_MSVC_RUNTIME_LIBRARY)
        if(crt)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "${crt}")
        endif()
        if(deploy)
            get_target_property(kind ${library} TYPE)
            if(NOT library STREQUAL target AND kind STREQUAL "SHARED_LIBRARY")
                list(APPEND files "$<TARGET_FILE:${library}>")
            endif()
            get_target_property(runtime_files ${library} CUDAVM_RUNTIME_FILES)
            if(runtime_files)
                list(APPEND files ${runtime_files})
            endif()
        endif()
        get_target_property(dependencies ${library} INTERFACE_LINK_LIBRARIES)
        if(dependencies)
            list(APPEND queue ${dependencies})
        endif()
        get_target_property(sdk_linkage ${library} INTERFACE_CUDAVM_LINKAGE)
        if(NOT sdk_linkage)
            get_target_property(dependencies ${library} LINK_LIBRARIES)
            if(dependencies)
                list(APPEND queue ${dependencies})
            endif()
        endif()
    endwhile()
    if(files)
        list(REMOVE_DUPLICATES files)
        add_custom_target(${target}_deploy
            COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:${target}>
            COMMAND ${CMAKE_COMMAND} -E copy_if_different ${files} $<TARGET_FILE_DIR:${target}>
            DEPENDS ${files} COMMENT "Deploying ${target} runtime libraries" VERBATIM)
        add_dependencies(${target} ${target}_deploy)
    endif()
endfunction()

function(cudavm_target_sources target)
    cmake_parse_arguments(ARG "DEVICE_LIBRARY;SELECTIVE" "ARCH;BACKEND;RUNTIME" "SOURCES;DEVICE_LIBRARIES;HOST_LIBRARIES" ${ARGN})
    if(NOT TARGET "${target}" OR (NOT ARG_SOURCES AND NOT ARG_DEVICE_LIBRARIES) OR ARG_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "cudavm_target_sources requires an existing target and SOURCES or DEVICE_LIBRARIES")
    endif()
    get_property(configured TARGET "${target}" PROPERTY CUDAVM_DEVICE_LINK_INPUTS SET)
    if(configured)
        message(FATAL_ERROR "Call cudavm_target_sources once per target with all mixed sources")
    endif()
    if(ARG_DEVICE_LIBRARY AND ARG_BACKEND)
        message(FATAL_ERROR "A DEVICE_LIBRARY retains translation-unit IR; select BACKEND on the final program")
    endif()
    if(NOT TARGET cudavm::nvcc)
        message(FATAL_ERROR "Mixed-source compilation requires the cudavm compiler SDK component")
    endif()
    set(compiler_dependencies cudavm::nvcc "$<TARGET_FILE:cudavm::core_shared>")
    if(NOT ARG_RUNTIME)
        set(ARG_RUNTIME cudavm::cudart_shared)
    endif()
    if(NOT ARG_ARCH)
        set(ARG_ARCH sm_75)
    endif()
    if(ARG_BACKEND AND NOT ARG_BACKEND MATCHES "^(cuda|opencl|opengl|vulkan|directx|all)$")
        message(FATAL_ERROR "Unknown cudavm backend ${ARG_BACKEND}")
    endif()
    get_target_property(standard "${target}" CUDA_STANDARD)
    if(NOT standard)
        get_target_property(standard "${target}" CXX_STANDARD)
    endif()
    if(NOT standard)
        set(standard 20)
    endif()
    set(root "${CMAKE_CURRENT_BINARY_DIR}/cudavm/${target}/$<CONFIG>")
    if(ARG_SELECTIVE AND NOT ARG_DEVICE_LIBRARY AND NOT ARG_SOURCES)
        file(GENERATE OUTPUT "${root}/link_root.cu" CONTENT "\n")
        list(APPEND ARG_SOURCES "${root}/link_root.cu")
    endif()
    set(inputs)
    target_link_libraries("${target}" PUBLIC ${ARG_RUNTIME})
    target_link_libraries("${target}" PRIVATE ${ARG_DEVICE_LIBRARIES})
    cudavm_prepare_target(${target})
    foreach(source IN LISTS ARG_SOURCES)
        get_filename_component(input "${source}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        get_filename_component(name "${source}" NAME_WE)
        get_filename_component(source_dir "${input}" DIRECTORY)
        string(SHA256 digest "${input}")
        string(SUBSTRING "${digest}" 0 12 digest)
        set(stem "${root}/${name}_${digest}")
        add_custom_command(OUTPUT "${stem}.host.cpp" "${stem}.bc" "${stem}.json"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${root}"
            COMMAND $<TARGET_FILE:cudavm::nvcc> compile-tu --input "${input}"
                --host "${stem}.host.cpp" --bitcode "${stem}.bc" --manifest "${stem}.json"
                --depfile "${stem}.d" --standard "c++${standard}" --arch "${ARG_ARCH}"
                "$<$<BOOL:$<TARGET_PROPERTY:${target},INCLUDE_DIRECTORIES>>:--include;$<JOIN:$<TARGET_PROPERTY:${target},INCLUDE_DIRECTORIES>,;--include;>>"
                "$<$<BOOL:$<TARGET_PROPERTY:${target},COMPILE_DEFINITIONS>>:--define;$<JOIN:$<TARGET_PROPERTY:${target},COMPILE_DEFINITIONS>,;--define;>>"
            DEPENDS "${input}" ${compiler_dependencies}
            DEPFILE "${stem}.d" COMMAND_EXPAND_LISTS VERBATIM)
        target_sources("${target}" PRIVATE "${stem}.host.cpp")
        if(CMAKE_CONFIGURATION_TYPES)
            set(configurations ${CMAKE_CONFIGURATION_TYPES})
        else()
            set(configurations "${CMAKE_BUILD_TYPE}")
        endif()
        foreach(config IN ITEMS ${configurations} "")
            string(REPLACE "$<CONFIG>" "${config}" host_source "${stem}.host.cpp")
            get_filename_component(host_source "${host_source}" ABSOLUTE)
            if(MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
                set_source_files_properties("${host_source}" PROPERTIES COMPILE_OPTIONS "/I${source_dir}")
            else()
                set_source_files_properties("${host_source}" PROPERTIES COMPILE_OPTIONS "-iquote;${source_dir}")
            endif()
        endforeach()
        list(APPEND inputs "${stem}.bc")
    endforeach()
    set_property(TARGET "${target}" PROPERTY CUDAVM_DEVICE_OWN_INPUTS "${inputs}")
    set(library_inputs)
    foreach(library IN LISTS ARG_DEVICE_LIBRARIES)
        if(NOT TARGET "${library}")
            message(FATAL_ERROR "Unknown device library: ${library}")
        endif()
        get_target_property(member_inputs "${library}" CUDAVM_DEVICE_LINK_INPUTS)
        if(NOT member_inputs)
            message(FATAL_ERROR "${library} does not expose cudavm device translation units")
        endif()
        get_target_property(library_kind "${library}" TYPE)
        if(ARG_SELECTIVE)
            get_target_property(member_inputs "${library}" CUDAVM_DEVICE_OWN_INPUTS)
            if(library_kind STREQUAL "STATIC_LIBRARY")
                list(APPEND library_inputs ${member_inputs})
            else()
                list(APPEND inputs ${member_inputs})
            endif()
        else()
            list(APPEND inputs ${member_inputs})
        endif()
    endforeach()
    list(REMOVE_DUPLICATES inputs)
    set_property(TARGET "${target}" PROPERTY CUDAVM_DEVICE_LINK_INPUTS "${inputs}")
    if(ARG_DEVICE_LIBRARY)
        add_custom_target("${target}_device" DEPENDS ${inputs} ${library_inputs})
        add_dependencies("${target}" "${target}_device")
        return()
    endif()
    foreach(input IN LISTS inputs)
        list(APPEND bitcode --bitcode "${input}")
    endforeach()
    set(host_dependencies)
    if(ARG_SELECTIVE)
        list(REMOVE_DUPLICATES library_inputs)
        foreach(input IN LISTS library_inputs)
            list(APPEND bitcode --library-bitcode "${input}")
        endforeach()
        get_target_property(host_sources ${target} SOURCES)
        set(host_target "${target}_cudavm_host")
        add_library(${host_target} OBJECT ${host_sources})
        foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES)
            set_property(TARGET ${host_target} PROPERTY ${property} "$<TARGET_PROPERTY:${target},${property}>")
        endforeach()
        foreach(property CXX_STANDARD CXX_STANDARD_REQUIRED CXX_EXTENSIONS MSVC_RUNTIME_LIBRARY
                POSITION_INDEPENDENT_CODE VISIBILITY_INLINES_HIDDEN CXX_VISIBILITY_PRESET
                C_VISIBILITY_PRESET C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS)
            get_target_property(value ${target} ${property})
            if(NOT value STREQUAL "value-NOTFOUND")
                set_property(TARGET ${host_target} PROPERTY ${property} "${value}")
            endif()
        endforeach()
        get_target_property(kind ${target} TYPE)
        if(kind STREQUAL "SHARED_LIBRARY" OR kind STREQUAL "MODULE_LIBRARY")
            set_property(TARGET ${host_target} PROPERTY POSITION_INDEPENDENT_CODE ON)
            get_target_property(export_define ${target} DEFINE_SYMBOL)
            if(NOT export_define)
                string(MAKE_C_IDENTIFIER "${target}_EXPORTS" export_define)
            endif()
            target_compile_definitions(${host_target} PRIVATE "${export_define}")
        endif()
        set_property(TARGET ${target} PROPERTY SOURCES "$<TARGET_OBJECTS:${host_target}>")
        set(host_command "--host-input;$<JOIN:$<TARGET_OBJECTS:${host_target}>,;--host-input;>")
        list(APPEND host_dependencies ${host_target} "$<TARGET_OBJECTS:${host_target}>")
        foreach(library IN LISTS ARG_HOST_LIBRARIES)
            get_target_property(library_kind ${library} TYPE)
            if(library_kind STREQUAL "OBJECT_LIBRARY")
                string(APPEND host_command ";--host-input;$<JOIN:$<TARGET_OBJECTS:${library}>,;--host-input;>")
                list(APPEND host_dependencies ${library} "$<TARGET_OBJECTS:${library}>")
            else()
                list(APPEND bitcode --host-input "$<TARGET_FILE:${library}>")
                list(APPEND host_dependencies ${library} "$<TARGET_FILE:${library}>")
            endif()
        endforeach()
    endif()
    set(outputs "${root}/program.bc" "${root}/program.ll" "${root}/program.json")
    set(emission)
    if(ARG_BACKEND)
        list(APPEND outputs "${root}/program.ptx" "${root}/program.module.cpp")
        list(APPEND emission --target "${ARG_BACKEND}" --binary "${root}/program.ptx" --module "${root}/program.module.cpp")
        target_sources("${target}" PRIVATE "${root}/program.module.cpp")
    endif()
    add_custom_command(OUTPUT ${outputs}
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${root}"
        COMMAND $<TARGET_FILE:cudavm::nvcc> link-device ${bitcode} "${host_command}"
            --portable "${root}/program.bc" --ir "${root}/program.ll" --manifest "${root}/program.json"
            --arch "${ARG_ARCH}" ${emission}
        DEPENDS ${inputs} ${library_inputs} ${host_dependencies} ${compiler_dependencies} COMMAND_EXPAND_LISTS VERBATIM)
    add_custom_target("${target}_device" DEPENDS ${outputs})
    add_dependencies("${target}" "${target}_device")
    set_property(TARGET "${target}" PROPERTY CUDAVM_DEVICE_BITCODE "${root}/program.bc")
endfunction()

function(_cudavm_mark_sources target)
    get_target_property(sources ${target} SOURCES)
    foreach(source IN LISTS sources)
        if(source MATCHES "\\.cu$")
            if(source MATCHES "\\$<")
                message(FATAL_ERROR "${target}: configuration-dependent .cu source paths are not supported yet")
            endif()
            set_source_files_properties("${source}" PROPERTIES HEADER_FILE_ONLY TRUE)
        endif()
    endforeach()
endfunction()

function(_cudavm_device_libraries target device_output host_output)
    get_target_property(queue ${target} LINK_LIBRARIES)
    get_target_property(sources ${target} SOURCES)
    foreach(source IN LISTS sources)
        if(source MATCHES "^\\$<TARGET_OBJECTS:([^>]+)>$")
            list(APPEND queue "${CMAKE_MATCH_1}")
        endif()
    endforeach()
    set(visited ${target})
    set(result)
    set(host_libraries)
    while(queue)
        list(POP_FRONT queue library)
        string(REGEX REPLACE "^\\$<LINK_ONLY:([^>]+)>$" "\\1" library "${library}")
        if(NOT TARGET "${library}")
            continue()
        endif()
        get_target_property(owner ${library} ALIASED_TARGET)
        if(owner)
            set(library ${owner})
        endif()
        if(library IN_LIST visited)
            continue()
        endif()
        list(APPEND visited ${library})
        get_target_property(automatic ${library} CUDAVM_AUTOMATIC)
        if(automatic)
            _cudavm_finalize_target(${library})
        endif()
        get_target_property(inputs ${library} CUDAVM_DEVICE_LINK_INPUTS)
        get_target_property(resolved ${library} CUDAVM_DEVICE_RESOLVED)
        if(resolved)
            continue()
        endif()
        get_target_property(kind ${library} TYPE)
        get_target_property(sdk_linkage ${library} INTERFACE_CUDAVM_LINKAGE)
        if((kind STREQUAL "STATIC_LIBRARY" OR kind STREQUAL "OBJECT_LIBRARY") AND NOT sdk_linkage)
            list(APPEND host_libraries ${library})
        endif()
        if(inputs)
            list(APPEND result ${library})
        endif()
        get_target_property(dependencies ${library} INTERFACE_LINK_LIBRARIES)
        if(dependencies)
            list(APPEND queue ${dependencies})
        endif()
    endwhile()
    set(${device_output} ${result} PARENT_SCOPE)
    set(${host_output} ${host_libraries} PARENT_SCOPE)
endfunction()

function(_cudavm_finalize_target target)
    get_target_property(done ${target} CUDAVM_FINALIZED)
    if(done)
        return()
    endif()
    get_target_property(visiting ${target} CUDAVM_FINALIZING)
    if(visiting)
        message(FATAL_ERROR "${target}: cyclic device-library dependencies are not supported")
    endif()
    set_property(TARGET ${target} PROPERTY CUDAVM_FINALIZING TRUE)
    _cudavm_mark_sources(${target})
    _cudavm_device_libraries(${target} libraries host_libraries)
    get_target_property(sources ${target} SOURCES)
    set(cuda_sources)
    foreach(source IN LISTS sources)
        if(source MATCHES "\\.cu$")
            list(APPEND cuda_sources "${source}")
        endif()
    endforeach()
    get_target_property(runtime ${target} CUDA_RUNTIME_LIBRARY)
    if(NOT runtime OR runtime STREQUAL "Shared")
        set(runtime_target cudavm::cudart_shared)
    elseif(runtime STREQUAL "Static")
        set(runtime_target cudavm::cudart_static)
    else()
        message(FATAL_ERROR "${target}: CUDA_RUNTIME_LIBRARY must be Static or Shared")
    endif()
    target_link_libraries(${target} PUBLIC ${runtime_target})
    get_target_property(kind ${target} TYPE)
    if(cuda_sources OR libraries)
        get_target_property(architectures ${target} CUDA_ARCHITECTURES)
        if(NOT architectures)
            set(architectures 75)
        endif()
        set(archs)
        foreach(arch IN LISTS architectures)
            string(REGEX REPLACE "-(real|virtual)$" "" arch "${arch}")
            string(REGEX REPLACE "^sm_" "" arch "${arch}")
            if(NOT arch MATCHES "^[0-9]+[a-z]?$")
                message(FATAL_ERROR "${target}: unsupported CUDA_ARCHITECTURES '${architectures}'; specify one numeric architecture")
            endif()
            list(APPEND archs "${arch}")
        endforeach()
        list(REMOVE_DUPLICATES archs)
        list(LENGTH archs count)
        if(NOT count EQUAL 1)
            message(FATAL_ERROR "${target}: multi-architecture fatbinaries are pending; specify one CUDA_ARCHITECTURES value")
        endif()
        get_target_property(standard ${target} CUDA_STANDARD)
        if(NOT standard)
            get_target_property(standard ${target} CXX_STANDARD)
        endif()
        if(NOT standard)
            set(standard 20)
        endif()
        if(NOT standard MATCHES "^(17|20|23)$")
            message(FATAL_ERROR "${target}: CUDA_STANDARD ${standard} is not supported by the current mixed-source frontend")
        endif()
        set_property(TARGET ${target} PROPERTY CUDA_STANDARD ${standard})
        target_compile_features(${target} PRIVATE cxx_std_${standard})
        get_property(explicit_resolve TARGET ${target} PROPERTY CUDA_RESOLVE_DEVICE_SYMBOLS SET)
        get_target_property(resolve ${target} CUDA_RESOLVE_DEVICE_SYMBOLS)
        if(NOT explicit_resolve)
            if(kind STREQUAL "STATIC_LIBRARY" OR kind STREQUAL "OBJECT_LIBRARY")
                set(resolve OFF)
            else()
                set(resolve ON)
            endif()
        endif()
        if(kind STREQUAL "EXECUTABLE" AND NOT resolve)
            message(FATAL_ERROR "${target}: executables require CUDA_RESOLVE_DEVICE_SYMBOLS ON")
        endif()
        if(resolve AND (kind STREQUAL "STATIC_LIBRARY" OR kind STREQUAL "OBJECT_LIBRARY"))
            message(FATAL_ERROR "${target}: CUDA_RESOLVE_DEVICE_SYMBOLS ON for static/object libraries is pending; defer resolution to the final executable/shared library")
        endif()
        set(device_options)
        if(resolve)
            list(APPEND device_options BACKEND cuda SELECTIVE HOST_LIBRARIES ${host_libraries})
        else()
            list(APPEND device_options DEVICE_LIBRARY SELECTIVE)
        endif()
        cudavm_target_sources(${target} ${device_options} ARCH sm_${archs} RUNTIME ${runtime_target}
            SOURCES ${cuda_sources} DEVICE_LIBRARIES ${libraries})
        set_property(TARGET ${target} PROPERTY CUDAVM_DEVICE_RESOLVED ${resolve})
    endif()
    cudavm_prepare_target(${target})
    set_property(TARGET ${target} PROPERTY CUDAVM_FINALIZED TRUE)
    set_property(TARGET ${target} PROPERTY CUDAVM_FINALIZING FALSE)
endfunction()

function(cudavm_prepare_target target)
    cmake_parse_arguments(ARG "NO_DEPLOY" "" "" ${ARGN})
    if(NOT TARGET ${target} OR ARG_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "cudavm_prepare_target requires an existing target and optional NO_DEPLOY")
    endif()
    if(ARG_NO_DEPLOY)
        set_property(TARGET ${target} PROPERTY CUDAVM_NO_DEPLOY TRUE)
    endif()
    get_property(scheduled TARGET ${target} PROPERTY CUDAVM_PREPARE_SCHEDULED)
    if(NOT scheduled)
        set_property(TARGET ${target} PROPERTY CUDAVM_PREPARE_SCHEDULED TRUE)
        cmake_language(EVAL CODE "cmake_language(DEFER CALL _cudavm_prepare_target [[${target}]])")
    endif()
endfunction()

function(cudavm_add_target target kind)
    if(kind STREQUAL "EXECUTABLE")
        add_executable(${target} ${ARGN})
    else()
        add_library(${target} ${kind} ${ARGN})
    endif()
    _cudavm_mark_sources(${target})
    set_property(TARGET ${target} PROPERTY CUDAVM_AUTOMATIC TRUE)
    foreach(property STANDARD STANDARD_REQUIRED EXTENSIONS ARCHITECTURES
            RUNTIME_LIBRARY SEPARABLE_COMPILATION RESOLVE_DEVICE_SYMBOLS)
        if(DEFINED CMAKE_CUDA_${property})
            set_property(TARGET ${target} PROPERTY CUDA_${property} "${CMAKE_CUDA_${property}}")
        endif()
    endforeach()
    cmake_language(EVAL CODE "cmake_language(DEFER CALL _cudavm_finalize_target [[${target}]])")
endfunction()
