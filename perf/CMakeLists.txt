cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cudavm_perf LANGUAGES NONE)
    message(FATAL_ERROR "Configure the cudavm root with CUDAVM_BUILD_PERF=ON; private benchmark helpers come from the parent build")
endif()

add_library(cudavm_perf_framework STATIC perf_framework.cpp)
target_compile_features(cudavm_perf_framework PUBLIC cxx_std_20)
target_include_directories(cudavm_perf_framework PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32 AND MSVC)
    set_property(TARGET cudavm_perf_framework PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
add_custom_target(cudavm_perf DEPENDS cudavm_perf_framework)

function(cudavm_add_perf_slice name)
    cmake_parse_arguments(slice "HOST_ONLY" "" "CASES;COMPONENTS;BACKENDS" ${ARGN})
    if(slice_UNPARSED_ARGUMENTS OR slice_KEYWORDS_MISSING_VALUES)
        message(FATAL_ERROR "Invalid performance slice arguments for ${name}")
    endif()
    if(NOT name MATCHES "^[a-z][a-z0-9_]*$")
        message(FATAL_ERROR "Performance slice names must use descriptive snake_case")
    endif()
    if(slice_HOST_ONLY AND slice_BACKENDS)
        message(FATAL_ERROR "HOST_ONLY performance slices do not select backends")
    endif()

    set(components ${slice_COMPONENTS})
    set(backends ${CUDAVM_ENABLED_BACKENDS})
    if(slice_HOST_ONLY)
        set(backends host)
    else()
        list(PREPEND components cuda)
        if(slice_BACKENDS)
            foreach(backend IN LISTS slice_BACKENDS)
                if(NOT backend MATCHES "^(cuda|vulkan|directx|opencl|opengl)$")
                    message(FATAL_ERROR "Unknown performance backend: ${backend}")
                endif()
            endforeach()
            set(backends)
            foreach(backend IN LISTS CUDAVM_ENABLED_BACKENDS)
                if(backend IN_LIST slice_BACKENDS)
                    list(APPEND backends "${backend}")
                endif()
            endforeach()
        endif()
    endif()
    set(libraries cudavm_perf_framework)
    foreach(component IN LISTS components)
        if(NOT TARGET cudavm::${component}_shared)
            message(FATAL_ERROR "Performance slice ${name} requires cudavm::${component}_shared")
        endif()
        list(APPEND libraries cudavm::${component}_shared)
    endforeach()

    set(fixture_root "${CMAKE_CURRENT_SOURCE_DIR}/${name}")
    set(generated_root "${CMAKE_CURRENT_BINARY_DIR}/${name}")
    set(fixture_cubin_size 0)
    foreach(kind cu ptx ll cubin)
        set(fixture_${kind} "")
        set(path "${fixture_root}/${name}.${kind}")
        if(EXISTS "${path}")
            if(kind STREQUAL "cubin")
                file(READ "${path}" fixture_cubin HEX)
                string(LENGTH "${fixture_cubin}" fixture_cubin_size)
                math(EXPR fixture_cubin_size "${fixture_cubin_size} / 2")
                string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," fixture_cubin "${fixture_cubin}")
            else()
                file(READ "${path}" fixture_${kind})
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
        endif()
    endforeach()
    configure_file("${CMAKE_CURRENT_FUNCTION_LIST_DIR}/perf_fixtures.hpp.in"
        "${generated_root}/perf_fixtures.hpp" @ONLY)

    set(target "cudavm_perf_${name}")
    add_executable(${target} "${fixture_root}/${name}.cpp")
    target_include_directories(${target} PRIVATE "${generated_root}")
    target_link_libraries(${target} PRIVATE ${libraries})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(WIN32 AND MSVC)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
    cudavm_prepare_target(${target})
    add_dependencies(cudavm_perf ${target})

    set(cases ${slice_CASES})
    if(NOT cases)
        set(cases default)
    endif()
    foreach(backend IN LISTS backends)
        set(environment "CUDAVM_BACKENDS=${backend}")
        if(slice_HOST_ONLY)
            set(environment "CUDAVM_BACKENDS=")
        endif()
        foreach(case IN LISTS cases)
            set(test "perf.${name}.${backend}.${case}")
            add_test(NAME "${test}" COMMAND ${target} "${backend}" "${case}")
            set_tests_properties("${test}" PROPERTIES
                LABELS "perf;${name};${backend};${case}"
                ENVIRONMENT "${environment}"
                SKIP_RETURN_CODE 77 TIMEOUT 300 RUN_SERIAL TRUE
                WORKING_DIRECTORY "${generated_root}")
        endforeach()
    endforeach()
endfunction()
