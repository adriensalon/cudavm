cmake_minimum_required(VERSION 3.24)
project(cudavm_sdk_consumer LANGUAGES C CXX)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(CUDAVM_USE_STATIC_LIBS OFF)
if(SDK_LINKAGE STREQUAL "static")
    set(CUDAVM_USE_STATIC_LIBS ON)
endif()
find_package(cudavm CONFIG REQUIRED COMPONENTS cuda cudart ${SDK_LINKAGE}
    OPTIONAL_COMPONENTS nvrtc nvvm nvjitlink nvfatbin nvptxcompiler
    PATHS "${SDK_PREFIX}/lib/cmake/cudavm" NO_DEFAULT_PATH)
file(REAL_PATH "${SDK_PREFIX}" sdk_root)
function(sdk_check_target component)
    get_target_property(imported cudavm::${component}_${SDK_LINKAGE} IMPORTED)
    get_target_property(alias cudavm::${component} ALIASED_TARGET)
    if(NOT imported OR NOT alias STREQUAL "cudavm::${component}_${SDK_LINKAGE}")
        message(FATAL_ERROR "Wrong imported/default target for ${component}")
    endif()
    get_target_property(includes cudavm::${component}_${SDK_LINKAGE} INTERFACE_INCLUDE_DIRECTORIES)
    foreach(path IN LISTS includes)
        file(REAL_PATH "${path}" resolved)
        cmake_path(IS_PREFIX sdk_root "${resolved}" NORMALIZE contained)
        if(NOT contained OR NOT EXISTS "${path}")
            message(FATAL_ERROR "SDK include path escapes its installation: ${path}")
        endif()
    endforeach()
endfunction()
sdk_check_target(cuda)
sdk_check_target(cudart)
get_target_property(runtime_alias cudavm::runtime ALIASED_TARGET)
if(NOT runtime_alias STREQUAL "cudavm::cudart_${SDK_LINKAGE}")
    message(FATAL_ERROR "Runtime compatibility alias selects wrong variant")
endif()
add_executable(sdk_consumer main.cpp c_headers.c)
target_compile_features(sdk_consumer PRIVATE cxx_std_20 c_std_11)
target_link_libraries(sdk_consumer PRIVATE cudavm::runtime cudavm::cuda)
foreach(component nvrtc nvvm nvjitlink nvfatbin nvptxcompiler)
    if(TARGET cudavm::${component}_${SDK_LINKAGE})
        sdk_check_target(${component})
        target_link_libraries(sdk_consumer PRIVATE cudavm::${component})
        string(TOUPPER "${component}" define)
        target_compile_definitions(sdk_consumer PRIVATE SDK_WITH_${define})
    endif()
endforeach()
cudavm_configure_runtime_target(sdk_consumer RUNTIME cudavm::cudart_${SDK_LINKAGE})
foreach(component cuda cudart core nvrtc nvvm nvjitlink nvfatbin nvptxcompiler)
    if(TARGET cudavm::${component}_${SDK_LINKAGE})
        cudavm_deploy_runtime(sdk_consumer RUNTIME cudavm::${component}_${SDK_LINKAGE})
    endif()
endforeach()
include(CTest)
add_test(NAME sdk_consumer COMMAND sdk_consumer)
set_tests_properties(sdk_consumer PROPERTIES ENVIRONMENT "CUDAVM_BACKENDS=" TIMEOUT 60)
