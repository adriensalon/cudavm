cmake_minimum_required(VERSION 3.24)
project(stock_cuda_client LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
option(FIXTURE_RDC "Exercise an ordinary CUDA static library and device linking" OFF)

if(FIXTURE_RDC)
    add_library(device_fixture STATIC kernel.cu helper.cu dead.cu)
    set_target_properties(device_fixture PROPERTIES CUDA_SEPARABLE_COMPILATION ON CUDA_STANDARD 17)
    add_executable(cuda_stock main.cpp)
    set_target_properties(cuda_stock PROPERTIES CUDA_SEPARABLE_COMPILATION ON LINKER_LANGUAGE CUDA)
    target_link_libraries(cuda_stock PRIVATE device_fixture)
else()
    add_executable(cuda_stock launch.cu)
    set_target_properties(cuda_stock PROPERTIES CUDA_STANDARD 17)
endif()

target_compile_features(cuda_stock PRIVATE cxx_std_17)
target_link_libraries(cuda_stock PRIVATE CUDA::cudart_static CUDA::cuda_driver)
