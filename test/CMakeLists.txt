cmake_minimum_required(VERSION 3.24)

function(cudavm_add_test_slice name)
    cmake_parse_arguments(slice "API_ONLY;HOST_ONLY;THREADS;SOURCE_ONLY;SOURCE_CHECK;SOURCE_LTO" "FIXTURE;ARCHITECTURE;MINIMUM_PTX;SOURCE_STANDARD" "FAMILY_CHECKS;CASES;HOST_CASES;COMPONENTS;BACKENDS;PROFILES" ${ARGN})
    if(slice_SOURCE_ONLY AND (slice_API_ONLY OR slice_HOST_ONLY))
        message(FATAL_ERROR "SOURCE_ONLY requires an execution slice")
    endif()
    if(slice_PROFILES AND (slice_API_ONLY OR slice_HOST_ONLY))
        message(FATAL_ERROR "PROFILES requires an execution slice")
    endif()
    if(slice_SOURCE_CHECK AND NOT slice_SOURCE_ONLY)
        message(FATAL_ERROR "SOURCE_CHECK requires SOURCE_ONLY")
    endif()
    if(slice_SOURCE_LTO AND NOT slice_SOURCE_ONLY)
        message(FATAL_ERROR "SOURCE_LTO requires SOURCE_ONLY")
    endif()
    if(slice_MINIMUM_PTX AND (NOT slice_SOURCE_CHECK OR NOT slice_MINIMUM_PTX MATCHES "^[1-9][0-9]+$"))
        message(FATAL_ERROR "MINIMUM_PTX requires SOURCE_CHECK and a numeric major/minor pair")
    endif()
    if(slice_HOST_CASES AND (NOT slice_API_ONLY OR slice_HOST_ONLY))
        message(FATAL_ERROR "HOST_CASES requires an API_ONLY slice with backend cases")
    endif()
    set(components ${slice_COMPONENTS})
    if(slice_HOST_ONLY)
        set(slice_API_ONLY TRUE)
    else()
        list(PREPEND components cuda)
    endif()
    if(NOT slice_API_ONLY)
        list(APPEND components nvrtc)
        if(NOT slice_SOURCE_ONLY)
            list(APPEND components nvvm nvjitlink)
        endif()
    endif()
    set(libraries)
    foreach(component IN LISTS components)
        if(NOT TARGET cudavm::${component}_shared)
            message(VERBOSE "Skipping public ${name} test: cudavm::${component}_shared is unavailable")
            return()
        endif()
        list(APPEND libraries cudavm::${component}_shared)
    endforeach()
    set(fixture_name "${name}")
    if(slice_FIXTURE)
        set(fixture_name "${slice_FIXTURE}")
    endif()
    set(fixture_root "${CMAKE_CURRENT_SOURCE_DIR}/${fixture_name}")
    set(generated_root "${CMAKE_CURRENT_BINARY_DIR}/${name}")
    if(NOT slice_API_ONLY)
        set(fixture_architecture 75)
        set(fixture_architecture_suffix "")
        set(fixture_source_standard 17)
        if(slice_SOURCE_STANDARD)
            if(NOT slice_SOURCE_STANDARD MATCHES "^(11|14|17|20|23)$")
                message(FATAL_ERROR "Execution SOURCE_STANDARD must be a supported C++ dialect")
            endif()
            set(fixture_source_standard "${slice_SOURCE_STANDARD}")
        endif()
        if(slice_ARCHITECTURE)
            if(NOT slice_ARCHITECTURE MATCHES "^([1-9][0-9]+)(a)?$")
                message(FATAL_ERROR "Execution ARCHITECTURE must be a numeric CUDA SM with optional a suffix")
            endif()
            set(fixture_architecture "${CMAKE_MATCH_1}")
            set(fixture_architecture_suffix "${CMAKE_MATCH_2}")
        endif()
        set(fixture_minimum_ptx 0)
        if(slice_MINIMUM_PTX)
            set(fixture_minimum_ptx "${slice_MINIMUM_PTX}")
        endif()
        foreach(kind cu ptx ll cubin)
            set(fixture_${kind} "")
        endforeach()
        set(fixture_cubin_size 0)
        set(fixture_kinds cu)
        if(NOT slice_SOURCE_ONLY)
            list(APPEND fixture_kinds ptx ll cubin)
        endif()
        foreach(kind IN LISTS fixture_kinds)
            if(kind STREQUAL "cubin")
                file(READ "${fixture_root}/${fixture_name}.cubin" fixture_cubin HEX)
                string(LENGTH "${fixture_cubin}" fixture_cubin_size)
                math(EXPR fixture_cubin_size "${fixture_cubin_size} / 2")
                string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," fixture_cubin "${fixture_cubin}")
            else()
                file(READ "${fixture_root}/${fixture_name}.${kind}" fixture_${kind})
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${fixture_root}/${fixture_name}.${kind}")
        endforeach()
        file(MAKE_DIRECTORY "${generated_root}")
        configure_file("${CMAKE_CURRENT_FUNCTION_LIST_DIR}/test_fixtures.hpp.in"
            "${generated_root}/test_fixtures.hpp" @ONLY)
    endif()
    set(target "cudavm_test_${name}")
    add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/${name}/${name}.cpp" "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/test_framework.cpp")
    if(slice_HOST_ONLY)
        target_compile_definitions(${target} PRIVATE CUDAVM_TEST_HOST_ONLY)
    endif()
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}")
    target_link_libraries(${target} PRIVATE ${libraries} ${CMAKE_DL_LIBS})
    foreach(component nvrtc nvjitlink)
        if(component IN_LIST components)
            string(TOUPPER "${component}" compiler_api)
            target_compile_definitions(${target} PRIVATE "CUDAVM_TEST_${compiler_api}")
        endif()
    endforeach()
    if(slice_THREADS AND NOT WIN32)
        target_link_libraries(${target} PRIVATE Threads::Threads)
    endif()
    if(NOT slice_API_ONLY)
        target_include_directories(${target} PRIVATE "${generated_root}")
        target_compile_definitions(${target} PRIVATE CUDAVM_TEST_EXECUTION)
        if(slice_SOURCE_ONLY)
            target_compile_definitions(${target} PRIVATE CUDAVM_TEST_SOURCE_ONLY)
        endif()
        if(slice_SOURCE_LTO)
            target_compile_definitions(${target} PRIVATE CUDAVM_TEST_SOURCE_LTO)
        endif()
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(WIN32 AND MSVC)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
    cudavm_prepare_target(${target})
    add_dependencies(cudavm_test ${target})
    if(slice_SOURCE_CHECK)
        set(source_check_target "${target}_source_check")
        add_executable(${source_check_target} "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/test_framework.cpp")
        target_compile_features(${source_check_target} PRIVATE cxx_std_20)
        target_compile_definitions(${source_check_target} PRIVATE CUDAVM_TEST_HOST_ONLY CUDAVM_TEST_SOURCE_CHECK CUDAVM_TEST_NVRTC)
        target_include_directories(${source_check_target} PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}" "${generated_root}")
        target_link_libraries(${source_check_target} PRIVATE cudavm::nvrtc_shared)
        set_target_properties(${source_check_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
        if(WIN32 AND MSVC)
            set_property(TARGET ${source_check_target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
        endif()
        cudavm_prepare_target(${source_check_target})
        add_dependencies(cudavm_test ${source_check_target})
        add_test(NAME "test.${name}.host.source" COMMAND ${source_check_target})
        set_tests_properties("test.${name}.host.source" PROPERTIES
            LABELS "test;${name};host;source;${slice_FAMILY_CHECKS}"
            ENVIRONMENT "CUDAVM_BACKENDS=cuda" TIMEOUT 120 RUN_SERIAL TRUE)
    endif()
    set(importers api)
    if(slice_CASES)
        set(importers ${slice_CASES})
    endif()
    set(timeout 120)
    if(NOT slice_API_ONLY)
        set(importers cu ptx cubin nvvm)
        if(slice_SOURCE_ONLY)
            set(importers cu)
        endif()
        set(timeout 300)
    endif()
    set(backends ${test_backends})
    if(slice_BACKENDS)
        set(backends)
        foreach(backend IN LISTS test_backends)
            if(backend IN_LIST slice_BACKENDS)
                list(APPEND backends "${backend}")
            endif()
        endforeach()
    endif()
    if(slice_HOST_ONLY)
        set(backends host)
    endif()
    foreach(backend IN LISTS backends)
        foreach(importer IN LISTS importers)
            set(case "test.${name}.${backend}")
            set(labels "test;${name};${backend};${slice_FAMILY_CHECKS}")
            set(arguments "${backend}")
            if(slice_HOST_ONLY)
                set(arguments)
            endif()
            if(NOT slice_API_ONLY OR slice_CASES)
                string(APPEND case ".${importer}")
                list(APPEND labels "${importer}")
                list(APPEND arguments "${importer}")
            endif()
            set(profiles __default)
            if(slice_PROFILES)
                set(profiles ${slice_PROFILES})
            endif()
            foreach(profile IN LISTS profiles)
                set(profile_case "${case}")
                set(profile_labels "${labels}")
                set(profile_arguments ${arguments})
                if(slice_PROFILES)
                    string(APPEND profile_case ".${profile}")
                    list(APPEND profile_labels "${profile}")
                    list(APPEND profile_arguments "${profile}")
                endif()
                add_test(NAME "${profile_case}" COMMAND ${target} ${profile_arguments})
                set_tests_properties("${profile_case}" PROPERTIES
                    LABELS "${profile_labels}"
                    ENVIRONMENT "CUDAVM_BACKENDS=${backend}"
                    SKIP_RETURN_CODE 77 TIMEOUT ${timeout} RUN_SERIAL TRUE)
                if(NOT slice_API_ONLY)
                    set_tests_properties("${profile_case}" PROPERTIES WORKING_DIRECTORY "${generated_root}")
                endif()
            endforeach()
        endforeach()
    endforeach()
    foreach(host_case IN LISTS slice_HOST_CASES)
        set(case "test.${name}.host.${host_case}")
        add_test(NAME "${case}" COMMAND ${target} "${host_case}")
        set_tests_properties("${case}" PROPERTIES
            LABELS "test;${name};host;${host_case};${slice_FAMILY_CHECKS}"
            ENVIRONMENT "CUDAVM_BACKENDS=" TIMEOUT 120 RUN_SERIAL TRUE)
    endforeach()
endfunction()

function(_cudavm_configure_program_image_fixtures target folder)
    foreach(kind cu ptx)
        file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module_loading/module_loading.${kind}" module_loading_${kind})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/module_loading/module_loading.${kind}")
    endforeach()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/program_image_roundtrip/program_image_roundtrip.ll" program_image_nvvm_text)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/program_image_roundtrip/program_image_roundtrip.ll")
    configure_file(module_loading/module_fixtures.hpp.in "${folder}/module_fixtures.hpp" @ONLY)
    configure_file(program_image_roundtrip/nvvm_image_fixture.hpp.in "${folder}/nvvm_image_fixture.hpp" @ONLY)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${folder}" "${CMAKE_CURRENT_SOURCE_DIR}/program_image_roundtrip")
endfunction()
