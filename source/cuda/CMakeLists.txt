set(backend_sources)
foreach(backend IN LISTS CUDAVM_ENABLED_BACKENDS)
    list(APPEND backend_sources "${backend}_backend.cpp")
endforeach()
cudavm_add_component(cuda driver_api.cpp procedure_address.cpp opengl_graphics.cpp registry.cpp host_memory.cpp
    shader_module.cpp shader_services.cpp shader_ptx.cpp ${backend_sources})
set(cudavm_install_targets ${cudavm_install_targets} PARENT_SCOPE)
foreach(linkage shared static)
    if(NOT TARGET cudavm_cuda_${linkage})
        continue()
    endif()
    foreach(backend IN LISTS CUDAVM_ENABLED_BACKENDS)
        string(TOUPPER "${backend}" name)
        target_compile_definitions(cudavm_cuda_${linkage} PRIVATE CUDAVM_ENABLE_${name}=1)
    endforeach()
    foreach(dependency OPENCL VULKAN)
        if(CUDAVM_ENABLE_${dependency})
            target_include_directories(cudavm_cuda_${linkage} SYSTEM PRIVATE "${CUDAVM_${dependency}_INCLUDE_DIR}")
        endif()
    endforeach()
    if(WIN32 AND CUDAVM_ENABLE_OPENGL)
        target_link_libraries(cudavm_cuda_${linkage} PRIVATE user32 gdi32)
    elseif(NOT WIN32)
        target_link_libraries(cudavm_cuda_${linkage} PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    endif()
endforeach()
