cudavm_add_component(nvvm nvvm_api.cpp)
set(cudavm_install_targets ${cudavm_install_targets} PARENT_SCOPE)
foreach(linkage shared static)
    if(TARGET cudavm_nvvm_${linkage})
        target_link_libraries(cudavm_nvvm_${linkage} PUBLIC cudavm::core_${linkage})
        if(APPLE)
            set_property(TARGET cudavm_nvvm_${linkage} PROPERTY INSTALL_RPATH "@loader_path")
        elseif(UNIX)
            set_property(TARGET cudavm_nvvm_${linkage} PROPERTY INSTALL_RPATH "$ORIGIN")
        endif()
    endif()
endforeach()
