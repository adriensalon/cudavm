add_executable(cudavm-nvcc main.cpp)
add_executable(cudavm::nvcc ALIAS cudavm-nvcc)
add_executable(cudavm::compiler ALIAS cudavm-nvcc) # compatibility alias
set_target_properties(cudavm-nvcc PROPERTIES EXPORT_NAME nvcc MSVC_RUNTIME_LIBRARY MultiThreaded)
get_target_property(compiler_linkage cudavm::nvrtc INTERFACE_CUDAVM_LINKAGE)
set_target_properties(cudavm-nvcc PROPERTIES CUDAVM_COMPILER_LINKAGE ${compiler_linkage}
    EXPORT_PROPERTIES CUDAVM_COMPILER_LINKAGE)
target_compile_features(cudavm-nvcc PRIVATE cxx_std_20)
target_link_libraries(cudavm-nvcc PRIVATE cudavm::nvrtc cudavm::nvjitlink)
if(WIN32)
    target_compile_definitions(cudavm-nvcc PRIVATE NDEBUG _ITERATOR_DEBUG_LEVEL=0)
endif()
if(UNIX)
    file(RELATIVE_PATH libraries_from_bin "/${CMAKE_INSTALL_BINDIR}" "/${CMAKE_INSTALL_LIBDIR}")
    set_property(TARGET cudavm-nvcc PROPERTY INSTALL_RPATH "$ORIGIN/${libraries_from_bin}")
endif()
cudavm_deploy_runtime(cudavm-nvcc RUNTIME cudavm::nvrtc)
cudavm_deploy_runtime(cudavm-nvcc RUNTIME cudavm::nvjitlink)
install(TARGETS cudavm-nvcc EXPORT cudavmTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
